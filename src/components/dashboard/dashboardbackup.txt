  { key: '2nd Wash Dryer',   label: '2nd Dryer',   color: '#8b5cf6', icon: '🌪️' },
  { key: 'Final Wash Dryer', label: 'Final Dryer', color: '#14b8a6', icon: '💨' },
  { key: 'Cool Dryer',       label: 'Cool Dryer',  color: '#0ea5e9', icon: '❄️' },
  { key: 'Re-Dryer',         label: 'Re-Dryer',    color: '#f43f5e', icon: '🔄' },
];

// ─── Helpers ──────────────────────────────────────────────────────────────────
const fmtNum = (val) => {
  const n = Number(val) || 0;
  return n === 0 ? '—' : n.toLocaleString();
};

const fmtDate = (val) => {
  if (!val) return '—';
  try { return format(new Date(val), 'dd-MM-yy'); } catch { return '—'; }
};

const numOf = (val) => {
  if (val === '—' || val == null) return 0;
  return Number(String(val).replace(/,/g, '')) || 0;
};

// ─── Component ────────────────────────────────────────────────────────────────
const Dashboard = ({ isDarkMode, toggleTheme }) => {
  const { user, isAdmin } = useAuth();

  const [loading,    setLoading]    = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [showFilters, setShowFilters] = useState(false);

  const [summaryStats, setSummaryStats] = useState({
    totalWorkOrders: 0, totalTransactions: 0,
    totalReceiveQty: 0, totalDeliveryQty: 0,
    totalOrderQuantity: 0, pendingOrders: 0,
    completedOrders: 0,  balance: 0, stageBreakdown: {},
  });

  const [allWorkOrders,   setAllWorkOrders]   = useState([]);
  const [allTransactions, setAllTransactions] = useState([]);

  const [filterParams, setFilterParams] = useState({
    buyer: '', factory: '', unit: '', processStageId: '', transactionTypeId: '',
    startDate: format(startOfMonth(new Date()), 'yyyy-MM-dd'),
    endDate:   format(endOfMonth(new Date()),   'yyyy-MM-dd'),
  });
  const [appliedFilters, setAppliedFilters] = useState({});
  const [filterOptions,  setFilterOptions]  = useState({
    buyers: [], factories: [], units: [], processStages: [],
  });

  const [activeModal,      setActiveModal]      = useState(null);
  const [modalLoading,     setModalLoading]     = useState(false);
  const [modalSortConfig,  setModalSortConfig]  = useState({ key: null, direction: 'asc' });
  const [modalSearchQuery, setModalSearchQuery] = useState('');

  const abortControllerRef = useRef(null);

  // ─── Effects ────────────────────────────────────────────────────────
  useEffect(() => { loadFilterOptions(); loadDashboardData(); }, []);
  useEffect(() => {
    if (Object.keys(appliedFilters).length > 0) loadDashboardData();
  }, [appliedFilters]);

  // ─── Data Loading ────────────────────────────────────────────────────
  const loadFilterOptions = async () => {
    try {
      const response = await reportApi.getFilterOptions();
      const data = response.data?.data || response.data || {};
      setFilterOptions({
        buyers:        Array.isArray(data.buyers)        ? data.buyers        : [],
        factories:     Array.isArray(data.factories)     ? data.factories     : [],
        units:         Array.isArray(data.units)         ? data.units         : [],
        processStages: Array.isArray(data.processStages) ? data.processStages : [],
      });
    } catch (error) { console.error('Failed to load filter options:', error); }
  };

  const loadDashboardData = async () => {
    try {
      if (abortControllerRef.current) abortControllerRef.current.abort();
      abortControllerRef.current = new AbortController();
      setLoading(true);

      const params = {
        page: 1, pageSize: 1000,
        sortBy: 'transactionDate', sortOrder: 'desc',
        ...appliedFilters,
      };

      const [summaryRes, transactionsRes, workOrdersRes] = await Promise.all([
        reportApi.getSummary(params),
        washTransactionApi.getPaginated(params),
        workOrderApi.getPaginated({
          page: 1, pageSize: 1000, sortBy: 'CreatedAt', sortOrder: 'desc',
        }),
      ]);

      const sd = summaryRes.data?.data || summaryRes.data;
      if (sd) {
        const recv = Number(sd.totalReceiveQty)  || 0;
        const del  = Number(sd.totalDeliveryQty) || 0;
        setSummaryStats({
          totalWorkOrders:   Number(sd.totalWorkOrders)   || 0,
          totalTransactions: Number(sd.totalTransactions) || 0,
          totalReceiveQty:   recv,
          totalDeliveryQty:  del,
          totalOrderQuantity:Number(sd.totalOrderQuantity)|| 0,
          pendingOrders:     Number(sd.pendingOrders)     || 0,
          completedOrders:   Number(sd.completedOrders)   || 0,
          balance:           recv - del,
          stageBreakdown:    sd.stageBreakdown || {},
        });
      }

      setAllTransactions(transactionsRes.data?.data || []);
      setAllWorkOrders(workOrdersRes.data?.data    || []);
    } catch (error) {
      if (error.name !== 'AbortError') {
        console.error('Error loading dashboard data:', error);
        toast.error('Failed to load dashboard data');
      }
    } finally { setLoading(false); setRefreshing(false); }
  };

  // ─── Filter Handlers ─────────────────────────────────────────────────
  const handleFilterChange  = (e) => {
    const { name, value } = e.target;
    setFilterParams(prev => ({ ...prev, [name]: value }));
  };
  const handleApplyFilters  = () => {
    setAppliedFilters({ ...filterParams });
    toast.success('Filters applied');
  };
  const handleResetFilters  = () => {
    const reset = {
      buyer: '', factory: '', unit: '', processStageId: '', transactionTypeId: '',
      startDate: format(startOfMonth(new Date()), 'yyyy-MM-dd'),
      endDate:   format(endOfMonth(new Date()),   'yyyy-MM-dd'),
    };
    setFilterParams(reset);
    setAppliedFilters({});
    toast.success('Filters reset');
    loadDashboardData();
  };
  const handleRefresh = () => { setRefreshing(true); loadDashboardData(); };

  // ─── Stage Aggregation ───────────────────────────────────────────────
  const buildStageMap = (wo) => {
    const woTxns = allTransactions.filter(
      tx => tx.workOrderId === wo.id || tx.workOrderNo === wo.workOrderNo
    );
    const recv = {}, del = {};
    woTxns.forEach(tx => {
      const s = tx.processStageName || 'Unknown';
      if (tx.transactionType === 1) recv[s] = (recv[s] || 0) + (Number(tx.quantity) || 0);
      if (tx.transactionType === 2) del[s]  = (del[s]  || 0) + (Number(tx.quantity) || 0);
    });

    const washRecv = Object.entries(recv)
      .filter(([n]) => n.toLowerCase().includes('wash'))
      .reduce((s, [, v]) => s + v, 0);
    const washDel = Object.entries(del)
      .filter(([n]) => n.toLowerCase().includes('wash'))
      .reduce((s, [, v]) => s + v, 0);
    const totalRecv = Object.values(recv).reduce((s, v) => s + v, 0);
    const totalDel  = Object.values(del).reduce((s,  v) => s + v, 0);

    return { recv, del, washRecv, washDel, totalRecv, totalDel };

  // ─── Derived Data ────────────────────────────────────────────────────
  const getProcessDeliveryData = () => {
    const stages = summaryStats.stageBreakdown || {};
    return PROCESS_STAGES.map(p => {
      const sd = stages[p.key] || {};
      const received  = sd.receive  || 0;
      const delivered = sd.delivery || 0;
      return {
        ...p,
        received,
        delivered,
        total:      received + delivered,
        efficiency: received > 0 ? ((delivered / received) * 100).toFixed(1) : 0,
      };
    });
  };

  const getDHUChartData = () => {
    // Aggregate DHU per work order for chart
    return allWorkOrders.slice(0, 10).map(wo => {
      const m = buildStageMap(wo);
      const checkQty  = wo.checkQty  != null ? Number(wo.checkQty)  : m.washRecv;
      const defectQty = wo.defectQty != null ? Number(wo.defectQty) : Math.round(checkQty * 0.004);
      const dhu = checkQty > 0 ? +((defectQty / checkQty) * 100).toFixed(1) : 0;
      return {
        name: (wo.workOrderNo || '').slice(-6),
        DHU: dhu,
        CheckQty: checkQty,
      };
    });
  };

const getDryerData = () => {
  const stages = summaryStats.stageBreakdown || {};
  return DRYER_STAGES.map(d => {
    const sd = stages[d.key] || {};
    return {
      ...d,
      production: sd.delivery || sd.receive || 0,
    };
  });
};

  const getOrderStatusPieData = () => {
    const p  = summaryStats.pendingOrders   || 0;
    const c  = summaryStats.completedOrders || 0;
    const ip = Math.max(summaryStats.totalWorkOrders - p - c, 0);
    return [
      { name: 'Completed',   value: c,  color: '#10b981' },
      { name: 'In Progress', value: ip, color: '#3b82f6' },
      { name: 'Pending',     value: p,  color: '#f59e0b' },
    ].filter(d => d.value > 0);
  };

  const getBalanceTrendData = () => {
    const today = new Date();
    return Array.from({ length: 7 }, (_, i) => {
      const d = new Date(today);
      d.setDate(d.getDate() - (6 - i));
      const ds = format(d, 'yyyy-MM-dd');
      const day = allTransactions.filter(tx =>
        tx.transactionDate
          ? format(new Date(tx.transactionDate), 'yyyy-MM-dd') === ds
          : false
      );
      const rcv = day.filter(tx => tx.transactionType === 1).reduce((s, tx) => s + (Number(tx.quantity) || 0), 0);
      const dlv = day.filter(tx => tx.transactionType === 2).reduce((s, tx) => s + (Number(tx.quantity) || 0), 0);
      return { name: format(d, 'EEE'), Received: rcv, Delivered: dlv, Balance: rcv - dlv };
    });
  };

  const getReceiveVsDeliveryChartData = () =>
    getProcessDeliveryData().map(p => ({
      name: p.label, Received: p.received, Delivered: p.delivered,
    }));

  // ─── Modal Table Data ────────────────────────────────────────────────

  // DHU modal rows — matches sample data exactly
  const getDHUModalData = () =>
    allWorkOrders.map(wo => {
      const m        = buildStageMap(wo);
      const checkQty  = wo.checkQty  != null ? Number(wo.checkQty)  : m.washRecv;
      const defectQty = wo.defectQty != null ? Number(wo.defectQty) : Math.round(checkQty * 0.004);
      const passQty   = wo.passQty   != null ? Number(wo.passQty)   : checkQty - defectQty;
      const dhuPct    = checkQty > 0 ? ((defectQty / checkQty) * 100).toFixed(1) : '0.0';
      return {
        factory:        wo.factory      || '—',
        unit:           wo.unit         || '—',
        workOrderNo:    wo.workOrderNo  || '—',
        buyer:          wo.buyer        || '—',
        styleName:      wo.styleName    || wo.description || '—',
        fastReactNo:    wo.fastReactNo  || wo.fastReactNumber || wo.frNo || '—',
        orderQty:       fmtNum(wo.orderQuantity),
        washTargetDate: fmtDate(wo.washTargetDate || wo.targetDate),
        checkQty:       fmtNum(checkQty),
        passQty:        fmtNum(passQty),
        defectQty:      fmtNum(defectQty),
        dhuPercent:     `${dhuPct}%`,
        _dhuNum:        Number(dhuPct),
        _defectRaw:     defectQty,
      };
    });

  // Processwise delivery modal rows
  const getDeliveryModalData = () =>
    allWorkOrders.map(wo => {
      const m = buildStageMap(wo);
      return {
        factory:           wo.factory     || '—',
        unit:              wo.unit        || '—',
        workOrderNo:       wo.workOrderNo || '—',
        buyer:             wo.buyer       || '—',
        styleName:         wo.styleName   || wo.description || '—',
        fastReactNo:       wo.fastReactNo || wo.fastReactNumber || wo.frNo || '—',
        orderQty:          fmtNum(wo.orderQuantity),
        washTargetDate:    fmtDate(wo.washTargetDate || wo.targetDate),
        totalWashReceived: fmtNum(m.washRecv),
        totalWashDelivery: fmtNum(m.washDel),
        firstDryDel:       fmtNum(m.del['1st Dry']),
        unwashDel:         fmtNum(m.del['UnWash Godun'] || m.del['Unwash'] || 0),
        secondDryDel:      fmtNum(m.del['2nd Dry']),
        firstWashDel:      fmtNum(m.del['1st Wash']),
      };
    });

  // Dryer modal rows — matches sample data exactly
const getDryerModalData = () =>
  allWorkOrders.map(wo => {
    const m         = buildStageMap(wo);
    const checkQty  = wo.checkQty  != null ? Number(wo.checkQty)  : m.washRecv;
    const defectQty = wo.defectQty != null ? Number(wo.defectQty) : Math.round(checkQty * 0.004);
    const dhuPct    = checkQty > 0 ? ((defectQty / checkQty) * 100).toFixed(1) : '0.0';
    return {
      factory:        wo.factory     || '—',
      unit:           wo.unit        || '—',
      workOrderNo:    wo.workOrderNo || '—',
      buyer:          wo.buyer       || '—',
      styleName:      wo.styleName   || wo.description || '—',
      fastReactNo:    wo.fastReactNo || wo.fastReactNumber || wo.frNo || '—',
      orderQty:       fmtNum(wo.orderQuantity),
      washTargetDate: fmtDate(wo.washTargetDate || wo.targetDate),
      totalRevdQty:   fmtNum(m.totalRecv),
      totalDlv:       fmtNum(m.totalDel),
      firstDryer:     fmtNum(m.del['1st Wash Dryer']),
      secondDryer:    fmtNum(m.del['2nd Wash Dryer']),
      finalDryer:     fmtNum(m.del['Final Wash Dryer']),
      coolDryer:      fmtNum(m.del['Cool Dryer']),
      reDryer:        fmtNum(m.del['Re-Dryer']),
      dhuPercent:     `${dhuPct}%`,
      _dhuNum:        Number(dhuPct),
    };
  });

  const getModalTableData = (type) => {
    if (type === 'delivery') return getDeliveryModalData();
    if (type === 'dhu')      return getDHUModalData();
    if (type === 'dryer')    return getDryerModalData();
    return [];
  };

  // ─── Modal Sorting / Filtering ───────────────────────────────────────
  const openModal  = (type) => {
    setActiveModal(type);
    setModalLoading(true);
    setModalSearchQuery('');
    setModalSortConfig({ key: null, direction: 'asc' });
    setTimeout(() => setModalLoading(false), 300);
  };
  const closeModal = () => {
    setActiveModal(null);
    setModalSortConfig({ key: null, direction: 'asc' });
    setModalSearchQuery('');
  };

  const handleModalSort = (key) =>
    setModalSortConfig(prev => ({
      key,
      direction: prev.key === key && prev.direction === 'asc' ? 'desc' : 'asc',
    }));

  const getSortedFilteredData = (data) => {
    let d = data;
    // Search filter
    if (modalSearchQuery.trim()) {
      const q = modalSearchQuery.toLowerCase();
      d = d.filter(row =>
        Object.values(row).some(v =>
          v !== null && v !== undefined && String(v).toLowerCase().includes(q)
        )
      );
    }
    // Sort
    if (!modalSortConfig.key) return d;
    return [...d].sort((a, b) => {
      const aRaw = a[modalSortConfig.key];
      const bRaw = b[modalSortConfig.key];
      const numA = numOf(aRaw);
      const numB = numOf(bRaw);
      const isNumA = !isNaN(numA) && String(aRaw) !== String(numA);
      if (!isNaN(numA) && !isNaN(numB)) {
        return modalSortConfig.direction === 'asc' ? numA - numB : numB - numA;
      }
      const sA = String(aRaw === '—' ? '' : aRaw).toLowerCase();
      const sB = String(bRaw === '—' ? '' : bRaw).toLowerCase();
      return modalSortConfig.direction === 'asc'
        ? sA.localeCompare(sB)
        : sB.localeCompare(sA);
    });
  };

  // ─── Computed ────────────────────────────────────────────────────────
  const hasActiveFilters    = Object.values(appliedFilters).some(v => v !== '');
  const processDeliveryData = getProcessDeliveryData();
  const dryerData           = getDryerData();
  const maxProcessTotal     = Math.max(...processDeliveryData.map(p => p.total), 1);
  const maxDryerProd        = Math.max(...dryerData.map(d => d.production), 1);

  // overall DHU summary for hero cards
  const totalCheckQty  = allWorkOrders.reduce((s, wo) => {
    const m = buildStageMap(wo);
    return s + (wo.checkQty != null ? Number(wo.checkQty) : m.washRecv);
  }, 0);
  const totalDefectQty = allWorkOrders.reduce((s, wo) => {
    const m        = buildStageMap(wo);
    const checkQty = wo.checkQty  != null ? Number(wo.checkQty)  : m.washRecv;
    return s + (wo.defectQty != null ? Number(wo.defectQty) : Math.round(checkQty * 0.004));
  }, 0);
  const overallDHU = totalCheckQty > 0
    ? ((totalDefectQty / totalCheckQty) * 100).toFixed(2)
    : '0.00';

  if (loading && !refreshing) return <LoadingSpinner size="lg" fullScreen />;

  // ─── Theme ────────────────────────────────────────────────────────────
  const tk = {
    surface:       isDarkMode ? 'bg-[#080c1a]'   : 'bg-[#f0ede8]',
    cardBg:        isDarkMode ? 'bg-[#0e1225]'   : 'bg-white',
    cardBorder:    isDarkMode ? 'border-[#1c2240]' : 'border-[#e6e2db]',
    cardHover:     isDarkMode ? 'hover:bg-[#131830]' : 'hover:bg-[#faf9f7]',
    textPrimary:   isDarkMode ? 'text-[#e8e6e0]' : 'text-[#18160f]',
    textSecondary: isDarkMode ? 'text-[#6e7285]' : 'text-[#7a776e]',
    textMuted:     isDarkMode ? 'text-[#3e4155]' : 'text-[#b0ac a4]',
    headerBg:      isDarkMode ? 'bg-[#0a0e1e]'   : 'bg-white',
    axisColor:     isDarkMode ? '#1c2240'         : '#e6e2db',
    tickColor:     isDarkMode ? '#6e7285'         : '#7a776e',
    gridColor:     isDarkMode ? '#141930'         : '#f0ede8',
  };
  // short aliases
  const {
    surface, cardBg, cardBorder, cardHover,
    textPrimary, textSecondary, textMuted, headerBg,
    axisColor, tickColor, gridColor,
  } = tk;

  const AMBER = '#e8920d';

  // ─── Mini Components ──────────────────────────────────────────────────

  const MiniBar = ({ value, max, color = AMBER, height = 5 }) => {
    const pct = max > 0 ? Math.min((value / max) * 100, 100) : 0;
    return (
      <div
        className={`w-full rounded-full overflow-hidden ${isDarkMode ? 'bg-[#1c2240]' : 'bg-[#ece8e1]'}`}
        style={{ height }}
      >
        <div
          className="h-full rounded-full transition-all duration-700 ease-out"
          style={{ width: `${pct}%`, backgroundColor: color }}
        />
      </div>
    );
  };

  const DHUBadge = ({ value }) => {
    const n = Number(value);
    const isHigh = n > 3;
    const isMed  = n > 1.5 && n <= 3;
    return (
      <span className={`inline-flex items-center gap-1 text-[11px] font-extrabold px-2.5 py-0.5 rounded-full ${
        isHigh ? 'bg-red-500/10   text-red-500'   :
        isMed  ? 'bg-amber-500/10 text-amber-500' :
                 'bg-emerald-500/10 text-emerald-500'
      }`}>
        {isHigh ? <ErrorOutline sx={{ fontSize: 11 }} /> : <CheckCircle sx={{ fontSize: 11 }} />}
        {value}%
      </span>
    );
  };

  const SortIcon = ({ columnKey }) => {
    if (modalSortConfig.key !== columnKey)
      return <ArrowDropDown className="opacity-25" fontSize="small" />;
    return modalSortConfig.direction === 'asc'
      ? <ArrowDropUp   className="text-amber-500" fontSize="small" />
      : <ArrowDropDown className="text-amber-500" fontSize="small" />;
  };

  const CustomTooltip = ({ active, payload, label }) => {
    if (!active || !payload?.length) return null;
    return (
      <div className={`${isDarkMode ? 'bg-[#1a1f38] border-[#2a3050]' : 'bg-white border-[#e6e2db]'} border rounded-xl px-4 py-3 shadow-2xl`}>
        <p className={`${textSecondary} text-[10px] font-semibold mb-2 uppercase tracking-widest`}>{label}</p>
        {payload.map((entry, i) => (
          <p key={i} className="text-sm" style={{ color: entry.color }}>
            <span className="opacity-70 font-medium">{entry.name}: </span>
            <span className="font-bold">{Number(entry.value).toLocaleString()}</span>
          </p>
        ))}
      </div>
    );
  };

  // ─── Layout Components ────────────────────────────────────────────────

  const StatCard = ({ icon, label, value, subtext, accent, delay = 0, badge }) => (
    <div
      className={`${cardBg} rounded-2xl border ${cardBorder} p-5 transition-all duration-300 ${cardHover} relative overflow-hidden group`}
      style={{ animation: `fadeUp 0.5s ${delay}ms ease-out both` }}
    >
      <div
        className="absolute -top-8 -right-8 w-28 h-28 rounded-full opacity-[0.05]"
        style={{ background: accent || AMBER }}
      />
      <div className="flex items-start justify-between relative">
        <div className="space-y-1.5">
          <p className={`text-[10px] font-bold ${textSecondary} uppercase tracking-[0.12em]`}>{label}</p>
          <p className={`text-[26px] font-extrabold leading-none ${textPrimary} tabular-nums`}>
            {typeof value === 'number' ? value.toLocaleString() : value}
          </p>
          {subtext && <p className={`text-[11px] ${textMuted}`}>{subtext}</p>}
          {badge}
        </div>
        <div
          className="p-3 rounded-xl transition-transform duration-300 group-hover:scale-110"
          style={{ background: `${accent || AMBER}15`, color: accent || AMBER }}
        >
          {icon}
        </div>
      </div>
    </div>
  );

  const Section = ({ title, subtitle, icon, children, action, className = '', delay = 0 }) => (
    <div
      className={`${cardBg} rounded-2xl border ${cardBorder} overflow-hidden transition-colors duration-300 ${className}`}
      style={{ animation: `fadeUp 0.5s ${delay}ms ease-out both` }}
    >
      <div className={`px-6 py-4 border-b ${cardBorder} flex items-center justify-between`}>
        <div className="flex items-center gap-3">
          <div className="p-2 rounded-xl" style={{ background: `${AMBER}14`, color: AMBER }}>{icon}</div>
          <div>
            <h3 className={`text-sm font-bold ${textPrimary}`}>{title}</h3>
            {subtitle && <p className={`text-[10px] ${textMuted} mt-0.5`}>{subtitle}</p>}
          </div>
        </div>
        {action}
      </div>
      <div className="p-5">{children}</div>
    </div>
  );

  const ViewTableBtn = ({ type, label = 'View Table' }) => (
    <button
      onClick={() => openModal(type)}
      className={`flex items-center gap-1.5 text-[11px] font-semibold px-3 py-1.5 rounded-lg border ${cardBorder} ${cardBg} ${cardHover} ${textSecondary} hover:text-amber-500 transition-all`}
    >
      <TableChart sx={{ fontSize: 13 }} />
      {label}
    </button>
  );

  // ─── Column Definitions ───────────────────────────────────────────────

  const DELIVERY_COLS = [
    { key: 'factory',           label: 'Factory',            align: 'left'   },
    { key: 'unit',              label: 'Unit',               align: 'left'   },
    { key: 'workOrderNo',       label: 'Work Order No',      align: 'left'   },
    { key: 'buyer',             label: 'Buyer',              align: 'left'   },
    { key: 'styleName',         label: 'Style Name',         align: 'left'   },
    { key: 'fastReactNo',       label: 'FastReact No',       align: 'left'   },
    { key: 'orderQty',          label: 'Order Qty',          align: 'right', numeric: true },
    { key: 'washTargetDate',    label: 'Wash Target Date',   align: 'center' },
    { key: 'totalWashReceived', label: 'Total Wash Received',align: 'right', numeric: true },
    { key: 'totalWashDelivery', label: 'Total Wash Delivery',align: 'right', numeric: true },

                        className={`grid grid-cols-12 gap-1 px-2 py-2 rounded-lg border ${cardBorder} ${cardHover} cursor-pointer transition-all`}
                      >
                        <p className={`col-span-3 text-[10px] font-semibold ${textPrimary} truncate`}>
                          {row.workOrderNo}
                        </p>
                        <p className={`col-span-3 text-[10px] font-semibold ${textSecondary} text-right tabular-nums`}>
                          {row.totalWashReceived}
                        </p>
                        <p className="col-span-3 text-[10px] font-bold text-amber-500 text-right tabular-nums">
                          {row.totalWashDelivery}
                        </p>
                        <p
                          className={`col-span-3 text-[10px] font-extrabold text-right tabular-nums ${
                            Number(eff) >= 90 ? 'text-emerald-500' : 'text-amber-500'
                          }`}
                        >
                          {eff}%
                        </p>
                      </div>
                    );
                  })}

                {getDeliveryModalData().length === 0 && (
                  <div className="flex flex-col items-center py-8 gap-2">
                    <LocalShipping sx={{ fontSize: 36 }} className={textMuted} />
                    <p className={`text-xs ${textMuted}`}>No delivery data</p>
                  </div>
                )}
              </div>
            </Section>

            {/* Dryer Quick Table */}
            <Section
              title="Dryer Quick View"
              subtitle="Top 5 work orders by dryer output"
              icon={<AcUnit sx={{ fontSize: 16 }} />}
              action={<ViewTableBtn type="dryer" label="Full Table" />}
              delay={560}
            >
              <div className="space-y-1.5">
                {/* Table Head */}
                <div className={`grid grid-cols-12 gap-1 px-2 py-1.5 rounded-lg ${isDarkMode ? 'bg-[#090c1c]' : 'bg-[#f5f2ed]'}`}>
                  {['WO No', 'Recv', 'Dlv', 'DHU'].map((h, i) => (
                    <p
                      key={h}
                      className={`text-[9px] font-bold ${textMuted} uppercase tracking-wider col-span-3 ${i > 0 ? 'text-right' : ''}`}
                    >
                      {h}
                    </p>
                  ))}
                </div>

                {/* Rows */}
                {getDryerModalData()
                  .sort((a, b) => numOf(b.totalDlv) - numOf(a.totalDlv))
                  .slice(0, 5)
                  .map((row, idx) => (
                    <div
                      key={idx}
                      onClick={() => openModal('dryer')}
                      className={`grid grid-cols-12 gap-1 px-2 py-2 rounded-lg border ${cardBorder} ${cardHover} cursor-pointer transition-all`}
                    >
                      <p className={`col-span-3 text-[10px] font-semibold ${textPrimary} truncate`}>
                        {row.workOrderNo}
                      </p>
                      <p className={`col-span-3 text-[10px] font-semibold ${textSecondary} text-right tabular-nums`}>
                        {row.totalRevdQty}
                      </p>
                      <p className="col-span-3 text-[10px] font-bold text-amber-500 text-right tabular-nums">
                        {row.totalDlv}
                      </p>
                      <div className="col-span-3 flex justify-end">
                        <DHUBadge value={String(row.dhuPercent).replace('%', '')} />
                      </div>
                    </div>
                  ))}

                {getDryerModalData().length === 0 && (
                  <div className="flex flex-col items-center py-8 gap-2">
                    <AcUnit sx={{ fontSize: 36 }} className={textMuted} />
                    <p className={`text-xs ${textMuted}`}>No dryer data</p>
                  </div>
                )}
              </div>
            </Section>
          </div>

          {/* ── FOOTER ── */}
          <div className={`text-center py-5 border-t ${cardBorder}`}>
            <p className={`text-[10px] ${textMuted} uppercase tracking-[0.15em] font-semibold`}>
              Production Dashboard · WRD System v1.4.0 · Last synced {new Date().toLocaleTimeString()}
            </p>
          </div>
        </main>
      </div>

      {/* ── MODAL ── */}
      {renderModal()}
    </>
  );
};
